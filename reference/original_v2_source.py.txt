import tkinter as tk

from tkinter import ttk, messagebox
import sqlite3
from datetime import datetime

DB = 'bifma_traders_v2.db'

def db():
    c=sqlite3.connect(DB); c.execute('PRAGMA foreign_keys=ON')
    c.executescript('''
    CREATE TABLE IF NOT EXISTS stores(id INTEGER PRIMARY KEY,code TEXT UNIQUE,name TEXT NOT NULL);
    CREATE TABLE IF NOT EXISTS units(id INTEGER PRIMARY KEY,name TEXT UNIQUE NOT NULL);
    CREATE TABLE IF NOT EXISTS products(id INTEGER PRIMARY KEY,code TEXT UNIQUE NOT NULL,name TEXT NOT NULL,unit TEXT DEFAULT 'PCS',sale_rate REAL DEFAULT 0,purchase_rate REAL DEFAULT 0,stock REAL DEFAULT 0,store_id INTEGER);
    CREATE TABLE IF NOT EXISTS accounts(id INTEGER PRIMARY KEY,code TEXT UNIQUE,name TEXT NOT NULL,account_type TEXT DEFAULT 'CUSTOMER',mobile TEXT,phone TEXT,address TEXT,balance REAL DEFAULT 0);
    CREATE TABLE IF NOT EXISTS sales(id INTEGER PRIMARY KEY,bill_no TEXT UNIQUE,sale_date TEXT,account_id INTEGER,store_id INTEGER,discount REAL DEFAULT 0,freight REAL DEFAULT 0,total REAL DEFAULT 0,received REAL DEFAULT 0,balance REAL DEFAULT 0);
    CREATE TABLE IF NOT EXISTS sale_items(id INTEGER PRIMARY KEY,sale_id INTEGER NOT NULL,product_id INTEGER NOT NULL,qty REAL,rate REAL,discount REAL DEFAULT 0,line_total REAL);
    CREATE TABLE IF NOT EXISTS purchases(id INTEGER PRIMARY KEY,bill_no TEXT UNIQUE,purchase_date TEXT,account_id INTEGER,store_id INTEGER,discount REAL DEFAULT 0,freight REAL DEFAULT 0,total REAL DEFAULT 0);
    CREATE TABLE IF NOT EXISTS purchase_items(id INTEGER PRIMARY KEY,purchase_id INTEGER NOT NULL,product_id INTEGER NOT NULL,qty REAL,rate REAL,discount REAL DEFAULT 0,line_total REAL);
    CREATE TABLE IF NOT EXISTS stock_ledger(id INTEGER PRIMARY KEY,product_id INTEGER,store_id INTEGER,entry_date TEXT,ref_type TEXT,ref_id INTEGER,qty_in REAL DEFAULT 0,qty_out REAL DEFAULT 0,note TEXT);
    ''')
    if c.execute('SELECT COUNT(*) FROM stores').fetchone()[0]==0:
        c.executemany('INSERT INTO stores(code,name) VALUES(?,?)',[('SHOP','SHOP'),('GODOWN1','GODOWN #1')])
    for u in ('PCS','CARTON','DOZEN','PACKET','KGS'): c.execute('INSERT OR IGNORE INTO units(name) VALUES(?)',(u,))
    c.commit(); return c

def fmt(x): return f'{float(x or 0):,.2f}'

    def trans(self,kind):
        self.clear(); title='Sales' if kind=='SALE' else 'Purchases'; self.heading(title,'Header + multiple item lines (master/detail)')
        c=db(); acc=c.execute('SELECT id,code||\' - \'||name FROM accounts WHERE account_type IN (?,?) ORDER BY name',('CUSTOMER','BOTH') if kind=='SALE' else ('SUPPLIER','BOTH')).fetchall(); pro=c.execute('SELECT id,code||\' - \'||name FROM products ORDER BY name').fetchall(); stores=c.execute('SELECT id,code||\' - \'||name FROM stores').fetchall();c.close()

        ttk.Combobox(l,textvariable=pv,values=[x[1] for x in pro],state='readonly',width=34).pack(side='left',padx=4);tk.Entry(l,textvariable=qv,width=9).pack(side='left',padx=4);tk.Entry(l,textvariable=rv,width=11).pack(side='left',padx=4);tk.Entry(l,textvariable=ld,width=9).pack(side='left',padx=4)
        tree=ttk.Treeview(self.main,columns=('product','qty','rate','disc','total'),show='headings',height=10)
        for col,txt in zip(tree['columns'],['Product','Qty','Rate','Discount','Line Total']):tree.heading(col,text=txt);tree.column(col,width=150)
        tree.pack(fill='both',expand=True,padx=22,pady=5);lab=tk.Label(self.main,text='Total: 0.00 | Balance: 0.00',bg='#f4f6f8',fg='#17365d',font=('Segoe UI',13,'bold'));lab.pack(anchor='e',padx=25,pady=8)
        def addline():
            try:
                if not pv.get():raise ValueError('Select a product.')

                    pr=c.execute("SELECT id FROM products WHERE code||' - '||name=?",(p,)).fetchone();
                    if not pr:continue
                    c.execute(f'INSERT INTO {table} ({"sale_id" if kind=="SALE" else "purchase_id"},product_id,qty,rate,discount,line_total) VALUES(?,?,?,?,?,?)',(tid,pr[0],float(q),float(r),float(d),float(lt)))

WHERE id=?',(delta,pr[0]));c.execute('INSERT INTO stock_ledger(product_id,store_id,entry_date,ref_type,ref_id,qty_in,qty_out,note) VALUES(?,?,?,?,?,?,?,?)',(pr[0],sid,date,ref,tid,max(delta,0),max(-delta,0),bill.get()))
